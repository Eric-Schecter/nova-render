cmake_minimum_required(VERSION 3.24)
project(NovaSDKConsumer LANGUAGES CXX)
find_package(Nova 0.1 CONFIG REQUIRED)
add_executable(nova_sdk_consumer main.cpp)
target_link_libraries(nova_sdk_consumer PRIVATE Nova::Render)
# Verification must not accidentally resolve libraries from the producer checkout/vcpkg.
if(NOVA_EXPECTED_SDK)
    foreach(dependency Nova::Render glfw volk::volk imgui::imgui)
        if(TARGET ${dependency})
            get_target_property(configurations ${dependency} IMPORTED_CONFIGURATIONS)
            foreach(configuration IN LISTS configurations)
                get_target_property(location ${dependency} IMPORTED_LOCATION_${configuration})
                if(location)
                    cmake_path(IS_PREFIX NOVA_EXPECTED_SDK "${location}" NORMALIZE inside_sdk)
                    if(NOT inside_sdk)
                        message(FATAL_ERROR "Dependency escaped SDK: ${dependency} = ${location}")
                    endif()
                endif()
            endforeach()
        endif()
    endforeach()
endif()
