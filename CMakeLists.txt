        add_test(NAME ${name} COMMAND ${CMAKE_COMMAND}
            "-DEXE=$<TARGET_FILE:nova_triangle>" "-DEXPECT=${expected}"
            "-DFAILURE_EXPECTED=${rejected}" "-DARGS=${ARGN}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_cli.cmake")
        set_tests_properties(${name} PROPERTIES RUN_SERIAL TRUE TIMEOUT 75)
    endfunction()

    function(nova_init_failure name api selection)
        if(selection STREQUAL "Metal")
            set(error "triangle[.]metallib")
        elseif(selection STREQUAL "D3D12")
            set(error "triangle[.]vert[.]dxil")
        else()
            set(error "triangle[.]vert[.]spv")
        endif()
        set(device_args)
        if(selection STREQUAL "D3D12")
            # Test shader I/O independently of the VM GPU's DXIL support.
            list(APPEND device_args --warp)
        endif()
        add_test(NAME ${name} COMMAND ${CMAKE_COMMAND}
            "-DEXE=$<TARGET_FILE:nova_triangle>" "-DEXPECT=${error}"
            -DFAILURE_EXPECTED=TRUE -DALLOW_SELECTION=TRUE "-DSELECTION=${selection}"
            "-DARGS=--api;${api};${device_args};--shader-dir;${CMAKE_CURRENT_BINARY_DIR}/tests/empty-shaders;--frames;1"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_cli.cmake")
        set_tests_properties(${name} PROPERTIES RUN_SERIAL TRUE TIMEOUT 75)
    endfunction()

            set(available TRUE)
            set(label Metal)
        elseif(api STREQUAL "dx12" AND WIN32)
            set(available TRUE)
            set(label D3D12)
        endif()
        if(available)
            nova_cli_test(api-${api} ${label} FALSE --api ${api} --frames 120 --resize-test)
            nova_init_failure(init-failure-${api} ${api} ${label})
        else()
            nova_cli_test(reject-${api} "unavailable in this build/platform" TRUE --api ${api} --frames 1)
        endif()
    endforeach()

            set(executable nova_scene_shadows_test)
        elseif(case STREQUAL "scene-render")
            set(executable nova_scene_render_test)
        elseif(case STREQUAL "environment")
            set(executable nova_environment_render_test)
        elseif(case STREQUAL "animation")
            set(executable nova_animation_render_test)
        else()
            if(NOT NOVA_ENABLE_UI)
                continue()
            endif()
            set(executable nova_parameter_panel_test)
        endif()
        if(NOT TARGET ${executable})
            continue()
        endif()
        set(asset "${PROJECT_SOURCE_DIR}/assets")
        if(case STREQUAL "environment")
            set(asset "${asset}/Environment/venice_sunset.nenv" "${asset}/DamagedHelmet/DamagedHelmet.glb")
        elseif(case STREQUAL "ui")
            set(asset "${asset}/DamagedHelmet/DamagedHelmet.glb")
        endif()
        add_test(NAME ${case}-dx12-hardware COMMAND ${executable} dx12-hardware ${asset})
        set_tests_properties(${case}-dx12-hardware PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/hardware" TIMEOUT 300)
    endforeach()
    if(NOVA_ENABLE_PATH_TRACING)
        add_test(NAME path-trace-dx12-hardware COMMAND nova_path_trace_test dx12-hardware "${PROJECT_SOURCE_DIR}/assets" "${CMAKE_BINARY_DIR}/hardware")
        add_test(NAME render-modes-dx12-hardware COMMAND nova_render_modes_test dx12-hardware "${PROJECT_SOURCE_DIR}/assets")
        set_tests_properties(path-trace-dx12-hardware render-modes-dx12-hardware PROPERTIES TIMEOUT 300 WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/hardware")
        set(hardware_apis dx12-hardware)
        if(NOVA_ENABLE_VULKAN)
            list(APPEND hardware_apis vulkan)
        endif()
        foreach(api IN LISTS hardware_apis)
            add_test(NAME mlt-${api}-gpu COMMAND nova_mlt_test ${api} "${PROJECT_SOURCE_DIR}/assets" "${CMAKE_BINARY_DIR}/hardware/${api}")
            add_test(NAME comparison-${api}-gpu COMMAND nova_comparison_test ${api} "${PROJECT_SOURCE_DIR}/assets")
            add_test(NAME glass-${api}-gpu COMMAND nova_glass_test ${api} "${PROJECT_SOURCE_DIR}/assets" "${CMAKE_BINARY_DIR}/hardware/glass-${api}.bin")
            add_test(NAME ao-${api}-gpu COMMAND nova_ao_test ${api} "${PROJECT_SOURCE_DIR}/assets")
            set_tests_properties(mlt-${api}-gpu comparison-${api}-gpu glass-${api}-gpu ao-${api}-gpu
                PROPERTIES TIMEOUT 600 WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/hardware")
        endforeach()
    endif()
endif()
