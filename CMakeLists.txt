cmake_minimum_required(VERSION 3.24)
option(NOVA_ENABLE_VULKAN "Build the Vulkan backend" ON)
# Select manifest dependencies before the vcpkg toolchain runs inside project().
set(VCPKG_MANIFEST_NO_DEFAULT_FEATURES ON)
if(NOVA_ENABLE_VULKAN)
    list(APPEND VCPKG_MANIFEST_FEATURES vulkan)
endif()
project(nova_triangle LANGUAGES CXX)
if(NOT NOVA_ENABLE_VULKAN AND NOT APPLE AND NOT WIN32)
    message(FATAL_ERROR "This platform requires NOVA_ENABLE_VULKAN=ON")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(glfw3 CONFIG REQUIRED)
if(NOVA_ENABLE_VULKAN)
    find_package(volk CONFIG REQUIRED)
endif()
include(cmake/Shaders.cmake)
add_library(nova_render STATIC render/src/Renderer.cpp)
if(NOVA_ENABLE_VULKAN)
    target_sources(nova_render PRIVATE render/src/backends/VulkanBackend.cpp)
    target_compile_definitions(nova_render PRIVATE NOVA_HAS_VULKAN)
    target_link_libraries(nova_render PRIVATE volk::volk)
endif()
if(APPLE)
    enable_language(OBJCXX)
    target_sources(nova_render PRIVATE render/src/backends/MetalBackend.mm)
    set_source_files_properties(render/src/backends/MetalBackend.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    set_target_properties(nova_render PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED ON)
    target_link_libraries(nova_render PRIVATE "-framework Metal" "-framework QuartzCore" "-framework Cocoa")
elseif(WIN32)
    target_sources(nova_render PRIVATE render/src/backends/D3D12Backend.cpp)
    target_compile_definitions(nova_render PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(nova_render PRIVATE d3d12 dxgi)
endif()
target_include_directories(nova_render PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/render/include")
target_link_libraries(nova_render PRIVATE glfw)
target_compile_definitions(nova_render PRIVATE VK_NO_PROTOTYPES GLFW_INCLUDE_NONE)
add_executable(nova_triangle examples/triangle/main.cpp)
if(APPLE)
    set_target_properties(nova_triangle PROPERTIES MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_GUI_IDENTIFIER "org.nova.triangle"
        MACOSX_BUNDLE_BUNDLE_NAME "Nova Triangle")
endif()
add_dependencies(nova_triangle triangle_shaders)
target_link_libraries(nova_triangle PRIVATE nova_render glfw)
target_compile_definitions(nova_triangle PRIVATE GLFW_INCLUDE_NONE
    TRIANGLE_SHADER_DIR="${SHADER_OUTPUT_DIR}")
foreach(target nova_render nova_triangle)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
    endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
    function(nova_cli_test name expected rejected)
        add_test(NAME ${name} COMMAND ${CMAKE_COMMAND}
            "-DEXE=$<TARGET_FILE:nova_triangle>" "-DEXPECT=${expected}"
            "-DFAILURE_EXPECTED=${rejected}" "-DARGS=${ARGN}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_cli.cmake")
        set_tests_properties(${name} PROPERTIES RUN_SERIAL TRUE TIMEOUT 75)
    endfunction()
    if(APPLE)
        set(default_api Metal)
    elseif(WIN32)
        set(default_api D3D12)
    else()
        set(default_api Vulkan)
    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tests/empty-shaders")
    function(nova_init_failure name api selection)
        if(selection STREQUAL "Metal")
            set(error "triangle[.]metallib")
        elseif(selection STREQUAL "D3D12")
            set(error "triangle[.]vert[.]dxil")
        else()
            set(error "triangle[.]vert[.]spv")
        endif()
        set(device_args)
        if(selection STREQUAL "D3D12")
            # Test shader I/O independently of the VM GPU's DXIL support.
            list(APPEND device_args --warp)
        endif()
        add_test(NAME ${name} COMMAND ${CMAKE_COMMAND}
            "-DEXE=$<TARGET_FILE:nova_triangle>" "-DEXPECT=${error}"
            -DFAILURE_EXPECTED=TRUE -DALLOW_SELECTION=TRUE "-DSELECTION=${selection}"
            "-DARGS=--api;${api};${device_args};--shader-dir;${CMAKE_CURRENT_BINARY_DIR}/tests/empty-shaders;--frames;1"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_cli.cmake")
        set_tests_properties(${name} PROPERTIES RUN_SERIAL TRUE TIMEOUT 75)
    endfunction()
    nova_init_failure(auto-init-failure auto ${default_api})
    nova_cli_test(auto-default ${default_api} FALSE --frames 120 --resize-test)
    nova_cli_test(auto-explicit ${default_api} FALSE --api auto --frames 120 --resize-test)
    foreach(api vulkan metal dx12)
        set(available FALSE)
        if(api STREQUAL "vulkan" AND NOVA_ENABLE_VULKAN)
            set(available TRUE)
            set(label Vulkan)
        elseif(api STREQUAL "metal" AND APPLE)
            set(available TRUE)
            set(label Metal)
        elseif(api STREQUAL "dx12" AND WIN32)
            set(available TRUE)
            set(label D3D12)
        endif()
        if(available)
            nova_cli_test(api-${api} ${label} FALSE --api ${api} --frames 120 --resize-test)
            nova_init_failure(init-failure-${api} ${api} ${label})
        else()
            nova_cli_test(reject-${api} "unavailable in this build/platform" TRUE --api ${api} --frames 1)
        endif()
    endforeach()
    if(WIN32)
        nova_cli_test(dx12-warp D3D12 FALSE --api dx12 --warp --frames 120 --resize-test)
        nova_cli_test(auto-warp D3D12 FALSE --warp --frames 120 --resize-test)
    else()
        nova_cli_test(reject-auto-warp "WARP requires the D3D12 backend" TRUE --warp --frames 1)
    endif()
    nova_cli_test(reject-vulkan-warp "WARP requires the D3D12 backend" TRUE --api vulkan --warp --frames 1)
    nova_cli_test(reject-metal-warp "WARP requires the D3D12 backend" TRUE --api metal --warp --frames 1)
    if(NOT WIN32)
        nova_cli_test(reject-dx12-warp "unavailable in this build/platform" TRUE --api dx12 --warp --frames 1)
    endif()
endif()
